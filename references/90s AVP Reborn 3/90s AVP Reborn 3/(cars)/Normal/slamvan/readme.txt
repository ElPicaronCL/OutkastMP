;-----------------------Installation-------------------------------

;Use IMGTool V2 and open gta3.img (located in the models folder) and replace the files in the gta3.img
;file with its corrosponding file from the ZIP archive.  ALWAYS MAKE BACKUPS!

;Now go to the Data folder and open CARCOLS.DAT in Notepad and replace the line for the Slamvan with this one:

slamvan, 3,104, 44,35, 0,36, 46,26, 13,36, 0,26, 116,53, 101,26, 1,77

;Next open HANDLING.DAT and replace the line for the Slamvan with this one:

SLAMVAN       1950.0    4712.5   4.0 	0.0 	0.0 	-0.3 	125 	0.85 	0.90 	0.53 	5 160.0 40.0 10.0 R P 	10.0 	0.52 	0 	28.0 	1.4 	0.15 	0.0 	0.3 	0.0 	0.5 	0.25 	0.36 	0.42 	40000 	40202040 	2110000 	2 	3 	0 	

;Lastly open VEHICLES.IDE and replace the line for the Slamvan with this one:

535, 	slamvan, 	slamvan, 	car, 		SLAMVAN, 	SLAMVAN,	null,	normal,	10,	  0,  0, -1,	0.77,	0.77,	   0
